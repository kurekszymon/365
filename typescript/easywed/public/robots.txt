User-agent: *

# Invite links are bearer tokens - anyone holding one can join the wedding, so
# a crawled and indexed claim URL hands access to whoever reads the index.
Disallow: /invite/

# Auth callbacks carry tokens in the URL fragment and query.
Disallow: /auth/

# Wedding ids are dynamic, so there is no prerendered HTML for them and they
# fall back to index.html - which is the landing, and says index,follow. Block
# the crawl rather than serve marketing copy under a wedding URL. /wedding/local
# is static and prerendered, so it is allowed through to be read as noindex.
Disallow: /wedding/
Allow: /wedding/local

# A venue peeking at a linked couple's seat map. Same reasoning as /wedding/:
# the id is dynamic, so there is no prerendered HTML and it falls back to the
# shell. Note this is the ONLY /crm path blocked - /crm and /venue are
# prerendered with noindex instead, for the reason spelled out below.
Disallow: /crm/wedding/

# Note what is deliberately NOT disallowed: /home, /login, /signup, /settings,
# /venue and /crm are prerendered with a noindex meta tag instead. Disallow and noindex do not
# stack - a blocked URL cannot be fetched, so the noindex is never read and
# Google keeps listing the bare URL from inbound links. That is exactly how
# /home, /login and /wedding/local ended up in the index with zero clicks.
# Letting these be crawled is what gets them removed. They render nothing
# without a session, so there is no content to leak.
